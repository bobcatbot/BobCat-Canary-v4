import discord
import json
import sqlite3
import time as PyTime
from modules import bot as v
from discord.ext import commands

con = sqlite3.connect('databases/suggestions.db')
cur = con.cursor()

class VoteButtons(discord.ui.View):
    def __init__(self, client):
        super().__init__(timeout=None)
        self.client = client
    
    @discord.ui.button(emoji="<:up:1025725255467159632>", style=discord.ButtonStyle.green, custom_id="upvote")
    async def upvote(self, button: discord.ui.Button, interaction: discord.Interaction):
        cur.execute("SELECT * FROM suggestions WHERE guild = ? AND message = ?", (int(interaction.guild.id), interaction.message.id))
        data = cur.fetchone()
        if data:
            guilds = data[0]
            message = data[1]
            upvotes = data[6]
            downvotes = data[7]

            guild = self.client.get_guild(int(guilds))
            channel = guild.get_channel(interaction.channel.id)
            msg = await channel.fetch_message(int(message))

            good = json.loads(upvotes)
            bad = json.loads(downvotes)
            
            # if str(interaction.user.id) == author:
            #     return await interaction.response.send_message("You can't vote on your own suggestion!", ephemeral=True)
            
            if str(interaction.user.id) in good:
                class RemoveVote(discord.ui.View):
                    def __init__(self):
                        super().__init__(timeout=10)
                    @discord.ui.button(label="Leave", style=discord.ButtonStyle.green, custom_id="upvote")
                    async def upvote(self, button: discord.ui.Button, interaction: discord.Interaction):
                        good.remove(f"{interaction.user.id}")
                        cur.execute("UPDATE suggestions SET good = ? WHERE guild = ? AND message = ?", (json.dumps(good), guild.id, msg.id))
                        con.commit()

                        await interaction.response.edit_message(content="You've successfully been removed from this suggestion's voting.", view=None)
                        for em in msg.embeds:
                            embed = em.to_dict()
                            embed["fields"][0]["value"] = f'> **Upvotes:** {len(good)}\n> **Downvotes:** {len(bad)}'
                        await msg.edit(embed=discord.Embed().from_dict(embed))
                return await interaction.response.send_message("You have already upvoted the suggestion", view=RemoveVote(), ephemeral=True)
            
            if not f"{interaction.user.id}" in good:
                await interaction.response.defer()
                
                if f"{interaction.user.id}" in bad:
                    bad.remove(f"{interaction.user.id}")
                    good.append(f"{interaction.user.id}")
                else:
                    good.append(f"{interaction.user.id}")

                for em in msg.embeds:
                    embed = em.to_dict()
                    embed["fields"][0]["value"] = f'> **Upvotes:** {len(good)}\n> **Downvotes:** {len(bad)}'
                await msg.edit(embed=discord.Embed().from_dict(embed), view=self)
        
                cur.execute("UPDATE suggestions SET bad = ? WHERE guild = ? AND message = ?", (json.dumps(bad), guild.id, msg.id))
                cur.execute("UPDATE suggestions SET good = ? WHERE guild = ? AND message = ?", (json.dumps(good), guild.id, msg.id))
                con.commit()

                # await interaction.response.send_message("Your upvote has been recorded!", ephemeral=True)
                return

    @discord.ui.button(emoji="<:info:1025432351418875975>", style=discord.ButtonStyle.gray, custom_id="suggestion_infomation")
    async def info(self, button: discord.ui.Button, interaction: discord.Interaction):
        cur.execute("SELECT good, bad FROM suggestions WHERE guild = ? AND message = ?", (interaction.guild.id, interaction.message.id))
        data = cur.fetchone()
        if data:
            good = json.loads(data[0])
            bad = json.loads(data[1])

            upvote = ""
            downvote = ""
            for user in good:
                member = self.client.get_user(int(user))
                upvote += f"{member.mention}" + "\n"
            
            for user in bad:
                member = self.client.get_user(int(user))
                downvote += f"{member.mention}" + "\n"
            
            embed = discord.Embed(title="Suggestion Information", color=v.blurple)
            embed.add_field(name=f"Upvoted [{len(good)}]", value=f'{"** **" if upvote == "" else upvote}', inline=False)
            embed.add_field(name=f"Downvoted [{len(bad)}]", value=f'{"** **" if downvote == "" else downvote}', inline=False)
            await interaction.response.send_message(embed=embed, ephemeral=True)

    @discord.ui.button(emoji="<:message:1029097871141769336>", style=discord.ButtonStyle.gray, custom_id="comment")
    async def comments(self, button: discord.ui.Button, interaction: discord.Interaction):
        discussions = v.dashboard(interaction.guild.id, "SuggestionDiscussions")
        if discussions == "false":
            return await interaction.response.send_message("Commenting is disabled", ephemeral=True)
        
        cur.execute("SELECT * FROM suggestions WHERE guild = ? AND message = ?", (interaction.guild.id, interaction.message.id))
        data = cur.fetchone()
        if data:
            guilds = data[0]
            message = data[1]
            comment = data[5]
            
            client = self.client
            guild = self.client.get_guild(int(guilds))
            channel = guild.get_channel(interaction.channel.id)
            msg = await channel.fetch_message(int(message))
            comments = json.loads(comment)

            class CommentBtn(discord.ui.View):
                def __init__(self):
                    super().__init__(timeout=30)
                @discord.ui.button(emoji="<:new_message:1029097911444832297>", style=v.ButtonStyle(interaction))
                async def button(self, button: discord.ui.Button, interaction: discord.Interaction):
                    class MyModal(discord.ui.Modal):
                        def __init__(self):
                            super().__init__(
                                discord.ui.InputText(label="Comment", style=discord.InputTextStyle.long, max_length=1000,),
                                title="New Comment"
                            )
                        async def callback(self, interaction: discord.Interaction):
                            await interaction.response.defer()

                            time = PyTime.time()
                            comments.append(
                                (f"{interaction.user.id}", f"{int(time)}", f"{self.children[0].value}")
                            )
                            cur.execute("UPDATE suggestions SET comments = ? WHERE guild = ? AND message = ?", (json.dumps(comments), guild.id, msg.id))
                            con.commit()

                            msgs = ""
                            for user, time, comment in comments:
                                member = client.get_user(int(user))
                                msgs += (
                                    f"**{member.name}** {member.mention} · <t:{time}:R>"
                                    f"\n{comment}"
                                ) + "\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n"
                            
                            embed = discord.Embed(
                                color=v.blurple, 
                                title=f"Suggestion {em_title} [{len(comments)}]",
                                description=f"{msgs}"
                            )
                            await interaction.edit_original_response(embed=embed)
                    await interaction.response.send_modal(MyModal())
            
            em_title = 'Comment' if len(comments) == 1 else 'Comments'
            if len(comments) == 0:
                embed = discord.Embed(
                    color=v.blurple, 
                    title=f"Suggestion {em_title} [{len(comments)}]",
                    description="No comments to show"
                )
                return await interaction.response.send_message(content=embed, view=CommentBtn(), ephemeral=True)
            
            msgs = ""
            for user, time, comment in comments:
                member = client.get_user(int(user))
                msgs += (
                    f"**{member.name}** {member.mention} · <t:{time}:R>"
                    f"\n{comment}"
                ) + "\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n"
            
            embed = discord.Embed(
                color=v.blurple, 
                title=f"Suggestion {em_title} [{len(comments)}]",
                description=f"{msgs}"
            )
            await interaction.response.send_message(embed=embed, view=CommentBtn(), ephemeral=True)

class SuggestSlash(commands.Cog):
    def __init__(self, client):
        self.client = client

    @commands.Cog.listener()
    async def on_ready(self):
        cur.execute("CREATE TABLE IF NOT EXISTS suggestions (guild TEXT, message TEXT, author TEXT, count TEXT, status TEXT, comments TEXT, good TEXT, bad TEXT)")
        con.commit()
        self.client.add_view(VoteButtons(self.client))

    suggestion = discord.SlashCommandGroup("suggestion", "Suggestions Commands", guild_ids=v.guild_ids)

    @commands.command()
    async def views(self, ctx):
        await ctx.send("click the buttons below", view=VoteButtons(self.client))
    
    @suggestion.command(description="Give your suggestions")
    @discord.option("suggestion", description="Your suggestion", required=True)
    async def suggest(self, ctx, *, suggestion):
        cur.execute("SELECT * FROM suggestions WHERE guild = ?", (ctx.guild.id,))
        data = cur.fetchall()
        count = len(data) + 1 if data else 1

        suggest = v.dashboard(ctx.guild.id, "SuggestionSuggestChannel")
        anonymous = v.dashboard(ctx.guild.id, "SuggestionAnonymous")

        sugg_em = discord.Embed(color=0x7289da,
            title=f"Suggestion #{count}",
            description=f"{suggestion}"
        )
        if anonymous == "false":
            sugg_em.set_author(icon_url=ctx.author.avatar.url, name=f"{ctx.author.name}")
        else:
            pass
        sugg_em.add_field(name="Votes", value="> **Upvotes:** 0\n> **Downvotes:** 0")
        sugg_em.add_field(name="Status", value=">>> This suggestion is waiting \nfor an official answer!")
        
        try:
            await ctx.respond("Suggestion added")

            channel = self.client.get_channel(int(suggest))
            msg = await channel.send(embed=sugg_em, view=VoteButtons(self.client))

            cur.execute(
                "INSERT INTO suggestions (guild, message, author, count, status, comments, good, bad) VALUES (?, ?, ?, ?, ?, ?, ?, ?)", 
                (ctx.guild.id, msg.id, ctx.author.id, count, "pending", "[]", "[]", "[]")
            )
            con.commit()
        except sqlite3.OperationalError:
            await ctx.respond("Somethink went wrong.. failed to add se", ephemeral=True)
    
    @suggestion.command(description="Marks a suggestion as approved")
    @discord.option("suggestion_id", description="Use the number next to the hash, not the message id", required=True)
    @discord.option("reason", description="Input reason", required=False)
    async def approve(self, ctx, *, suggestion_id, reason=None):
        if not ctx.author.guild_permissions.manage_guild:
            return await ctx.respond("Missing `Manage Server` permissions", ephemeral=True)

        messageEditDecision = v.dashboard(ctx.guild.id, "SuggestionEditDecision")
        suggestChannel = v.dashboard(ctx.guild.id, "SuggestionSuggestChannel")
        decisionChannel = v.dashboard(ctx.guild.id, "SuggestionDecisionChannel")
        dmDecision = v.dashboard(ctx.guild.id, "SuggestionDmDecision")
        
        reason = "No reason given" if not reason else reason
        
        cur.execute("SELECT * FROM suggestions WHERE guild = ? AND count = ? ", (ctx.guild.id, suggestion_id))
        data = cur.fetchone()
        if data:
            message = data[1]
            author = data[2]
            
            dChannel = self.client.get_channel(int(decisionChannel))
            channel = self.client.get_channel(int(suggestChannel))
            message = await channel.fetch_message(int(message))
            
            cur.execute("UPDATE suggestions SET status = ? WHERE guild = ?", ("approved", ctx.guild.id))
            con.commit()
            
            for em in message.embeds:
                embed = em.to_dict()
                embed["fields"][1]["value"] = f'**Approved**\n> {reason}'
            emb = discord.Embed().from_dict(embed)
            
            await ctx.respond("Successfully marked suggestion as approved!")
            
            if messageEditDecision == "true":
                await message.edit(embed=emb, view=None)
            else:
                await dChannel.send(embed=emb)
            
            if dmDecision == "true":
                user = self.client.get_user(int(author))
                await user.send(
                    content=f"A suggestion that was made by you has been approved by {ctx.author}! \nReason: {reason}", 
                    embed=emb
                )
        else:
            return await ctx.respond("That is not a valid suggestion ID!", ephemeral=True)

    @discord.option("status", description="Marks a suggestion as considered, implemented", options=["Considered", "Implemented"], required=True)

    async def update(self, ctx, *, suggestion_id, status, reason=None):
