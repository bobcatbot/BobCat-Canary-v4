class YoutubeNotifications(commands.Cog):
    def __init__(self, client):
        self.client = client
        self.youtube = build('youtube', 'v3', developerKey=os.getenv('YOUTUBE_API_KEY')) # Initialize YouTube API client

        self.check_new_video.start()
    
    @tasks.loop(minutes=5)
    async def check_new_video(self):
        with open("server.json", "r", encoding='utf-8') as f:
            jdata = json.load(f)
        
        data = jdata['Bot']['social_alerts']['youtube']
        config = jdata['Dash']['social_alerts']['youtube']

        # Store the last video ID
        last_video_id = data['last_vid']

        # Fetch the latest video from the channel
        request = self.youtube.search().list(
            part='snippet',
            channelId=config['channel']['id'],
            maxResults=1,
            order='date',
            type='video',
            # eventType='live'
        )
        response = request.execute()

        if response['items']:
            latest_video = response['items'][0]

            video_id = latest_video['id']['videoId']
            channel_name = config['channel']['name']
            video_title = latest_video['snippet']['title']
            video_publishAt = latest_video['snippet']['publishedAt']

            video_url = f'https://www.youtube.com/watch?v={video_id}'

            # Check if the video is new
            if video_id != last_video_id:
                # Send a message to the Discord channel
                print("Hey, **{channel_name}** uploaded a new video! Go check it out! \n{video_url}".format(channel_name=channel_name, video_url=video_url, video_title=video_title))

                # Update the last video ID
                data['last_vid'] = video_id
                data['last_vurl'] = video_url
                data['published'] = video_publishAt

                with open("server.json", "w", encoding='utf-8') as f:
                    json.dump(jdata, f, indent=2)

TWITCH_CLIENT_SECRET = os.getenv("TWITCH_CLIENT_SECRET")
class TwitchNotifications(commands.Cog):
    def __init__(self, client):
        self.client = client
        self.access_token = None
        self.expires_at = None
        self.last_refresh_time = 0  # Timestamp of the last refresh

        self.check_new_stream.start()

    def is_token_valid(self):
        """Check if the cached token is still valid."""
        return self.access_token and self.expires_at and time.time() < self.expires_at
    
    def set_token(self, access_token, expires_in):
        """Cache the access token and its expiration time."""
        self.access_token = access_token
        self.expires_at = time.time() + expires_in - 60  # Subtract 60 seconds for buffer
        self.last_refresh_time = time.time()

    def get_token(self):
        """Get the cached access token."""
        return self.access_token

    def is_daily_refresh_needed(self):
        """Check if a daily refresh is needed."""
        return time.time() - self.last_refresh_time >= 86400  # 86400 seconds = 1 day

    async def get_twitch_access_token(self):# Check if the cached token is still valid
        if self.is_token_valid() and not self.is_daily_refresh_needed():
            return self.get_token()
    
        response = requests.post("https://id.twitch.tv/oauth2/token", params={
            "client_id": TWITCH_CLIENT_ID,
            "client_secret": TWITCH_CLIENT_SECRET,
            "grant_type": "client_credentials"
        }).json()

        # Cache the new token
        access_token = response["access_token"]
        expires_in = response["expires_in"]  # Token lifetime in seconds
        self.set_token(access_token, expires_in)

        return access_token

    @tasks.loop(minutes=2)
    async def check_new_stream(self):
        with open("server.json", "r", encoding='utf-8') as f:
            jdata = json.load(f)
        
        data = jdata['Bot']['social_alerts']['twitch']
        config = jdata['Dash']['social_alerts']['twitch']

        twitch_access_token = await self.get_twitch_access_token()

        headers = {
            "Client-ID": TWITCH_CLIENT_ID,
            "Authorization": f"Bearer {twitch_access_token}"
        }
        params = {
            "user_id": config['channel']['id']
        }

        response = requests.get("https://api.twitch.tv/helix/streams", headers=headers, params=params).json()
        is_live = bool(response["data"])

        if is_live and not data['is_live']: # if stream is live and last stream status was not live
            streamer = response['data'][0]['user_name']
            link = f'https://twitch.tv/{streamer}'
            title = response['data'][0]['title']

            # Send a message to the Discord channel
            print(f"Hey, **{streamer}** is live! Go check it out! \n{link}".format(link=link, streamer=streamer, title=title))

            # Update the last video ID
            data['last_id'] = response['data'][0]['id']
            data['is_live'] = is_live

            with open("server.json", "w", encoding='utf-8') as f:
                json.dump(jdata, f, indent=2)

        elif not is_live: # if stream is not live and last stream status was live
            data['is_live'] = False

            with open("server.json", "w", encoding='utf-8') as f:
                json.dump(jdata, f, indent=2)

@app.route("/logintoyoutube/<guild_id>")
def logintoyoutube(guild_id):
  # Step 1: Redirect the user to the OAuth authorization URL
  flow = Flow.from_client_secrets_file(
    CLIENT_SECRETS_FILE,
    scopes=SCOPES,
    redirect_uri=YTREDIRECT_URI
  )
  authorization_url, state = flow.authorization_url(
    access_type="offline",  # Request a refresh token
    include_granted_scopes="true",
    state=guild_id
  )

  # Store the state in the session for later validation
  session["state"] = state
  return redirect(authorization_url)

  # Extract the authorization code and state from the query parameters
  authorization_code = request.args.get("code")
  state = request.args.get("state")

  if not authorization_code:
    return "Authorization failed: No authorization code found.", 400

  # Exchange the authorization code for an access token

    'client_key': TIKTOK_CLIENT_KEY,
