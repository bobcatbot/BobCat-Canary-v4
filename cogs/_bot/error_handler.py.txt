import sys
import traceback
import discord
from discord.ext import commands

ERROR_LOG_CHANNEL_ID = 1110277292124536953

class ErrorLogger(commands.Cog):
    def __init__(self, bot):
        self.bot = bot

    async def send_error(self, title: str, error: Exception):
        """Send an embed with traceback to the error log channel."""
        tb = "".join(traceback.format_exception(type(error), error, error.__traceback__))
        cleaned_tb = "\n".join(
            line for line in tb.splitlines() if not set(line.strip()) == {"^"}
        )

        embed = discord.Embed(
            title=f"⚠️ {title}",
            description=f"**Error:** `{type(error).__name__}`",
            color=discord.Color.red()
        )
        embed.add_field(name="Traceback", value=f"```{cleaned_tb[:1018]}```", inline=False)
        embed.set_footer(text="Auto-generated error report")

        channel = self.bot.get_channel(ERROR_LOG_CHANNEL_ID)
        if channel:
            await channel.send(embed=embed)
        else:
            print("Error log channel not found.")

        print(tb)

    # Catch command errors (including cogs)
    @commands.Cog.listener()
    async def on_command_error(self, ctx, error):
        await self.send_error(f"Command Error in {ctx.command}", error)
    
    @commands.Cog.listener()
    async def on_application_command_error(self, ctx, error):
        await self.send_error(f"Command Error in {ctx.command}", error)

    # Catch all event errors
    @commands.Cog.listener()
    async def on_error(self, event_method, *args, **kwargs):
        _, error, tb = sys.exc_info()
        await self.send_error(f"Event Error in {event_method}", error)

def setup(bot):
    bot.add_cog(ErrorLogger(bot))
