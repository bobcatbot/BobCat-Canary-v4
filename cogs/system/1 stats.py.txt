import discord
from discord.ext import commands, tasks
from datetime import datetime, timedelta
from modules import bot as v

class Stats(commands.Cog):
    def __init__(self, client):
        self.client: commands.Bot = client

        # Dispatch dict avoids repetitive if-blocks (pattern from file 1).
        # Values are keys into the `stats` dict returned by calculate_statistics.
        self.COUNTER_HANDLERS = {
            "botCount":       lambda s: s["members"]["bots"],
            "humanCount":     lambda s: s["members"]["humans"],
            "onlineCount":    lambda s: s["members"]["online"],
            "totalMembers":   lambda s: s["members"]["total"],
            "textCount":      lambda s: s["channels"]["text"],
            "voiceCount":     lambda s: s["channels"]["voice"],
            "categoryCount":  lambda s: s["channels"]["categories"],
            "totalChannels":  lambda s: s["channels"]["total"],
            "roleCount":      lambda s: s["roles"]["total"],
        }

        # Guilds flagged for refresh by event listeners; drained by the loop.
        self.dirty_guilds: set[int] = set()

        # Per-channel cooldown tracking, since Discord allows ~2 name edits
        # per 10 minutes per channel. Keyed by channel id -> last edit time.
        self.last_edit: dict[int, datetime] = {}
        self.EDIT_COOLDOWN = timedelta(minutes=10)

    @commands.Cog.listener()
    async def on_ready(self):
        if not self.refresh_loop.is_running():
            self.refresh_loop.start()

    def cog_unload(self):
        self.refresh_loop.cancel()

    # --- Event listeners: mark dirty instead of writing immediately ---
    # This is the fix for file 2's bug: writing on every single event blew
    # through Discord's channel-rename rate limit almost instantly on active
    # servers. Marking dirty and batching through the loop avoids that.

    @commands.Cog.listener()
    async def on_member_join(self, member: discord.Member):
        self.dirty_guilds.add(member.guild.id)

    @commands.Cog.listener()
    async def on_member_remove(self, member: discord.Member):
        self.dirty_guilds.add(member.guild.id)

    @commands.Cog.listener()
    async def on_guild_channel_create(self, channel: discord.abc.GuildChannel):
        self.dirty_guilds.add(channel.guild.id)

    @commands.Cog.listener()
    async def on_guild_channel_delete(self, channel: discord.abc.GuildChannel):
        self.dirty_guilds.add(channel.guild.id)

    @commands.Cog.listener()
    async def on_guild_role_create(self, role: discord.Role):
        self.dirty_guilds.add(role.guild.id)

    @commands.Cog.listener()
    async def on_guild_role_delete(self, role: discord.Role):
        self.dirty_guilds.add(role.guild.id)

    # --- Core logic ---

    def calculate_statistics(self, guild: discord.Guild) -> dict:
        """Calculate members/channels/roles statistics for a guild."""
        return {
            "members": {
                "total":  len(guild.members),
                "humans": sum(1 for m in guild.members if not m.bot),
                "bots":   sum(1 for m in guild.members if m.bot),
                # matches file 1's definition: anything not offline counts
                # as "online" (includes idle/dnd). Flagged above — confirm.
                "online": sum(1 for m in guild.members if m.status != discord.Status.offline),
            },
            "channels": {
                "total":      len(guild.channels),
                "text":       len(guild.text_channels),
                "voice":      len(guild.voice_channels),
                "categories": len(guild.categories),
            },
            "roles": {
                "total": len(guild.roles),
            },
        }

    async def update_guild_stats(self, guild: discord.Guild, force: bool = False):
        """Refresh counter channels for a single guild from its Dash.stats config."""
        
        dash = v.db.get_dash(guild.id)
        if not dash:
            return

        stats_cfg = dash.get("stats")
        if not stats_cfg or not stats_cfg.get("enabled", False):
            return

        stats = self.calculate_statistics(guild)
        counters = stats_cfg.get("counters", [])
        now = datetime.now()
        changed = False

        for counter in counters:
            target = counter.get("target")
            handler = self.COUNTER_HANDLERS.get(target)
            if handler is None:
                continue

            channel_id = counter.get("channel_id")
            if not channel_id:
                continue

            channel = self.client.get_channel(int(channel_id))
            if channel is None:
                continue

            # Respect the rename cooldown unless this is a forced/first update.
            last = self.last_edit.get(channel.id)
            if not force and last and (now - last) < self.EDIT_COOLDOWN:
                continue

            count = handler(stats)
            counter["count"] = count
            new_name = counter["text"].format(count=count)

            if channel.name == new_name:
                continue

            await channel.edit(name=new_name)
            self.last_edit[channel.id] = now
            changed = True

        if changed:
            v.db.update_dash(guild.id, "stats.counters", counters)

    @tasks.loop(minutes=10)
    async def refresh_loop(self):
        dirty = self.dirty_guilds
        self.dirty_guilds = set()

        for guild in self.client.guilds:
            force = guild.id in dirty
            await self.update_guild_stats(guild, force=force)

    @refresh_loop.before_loop
    async def before_refresh_loop(self):
        await self.client.wait_until_ready()

def setup(client):
    client.add_cog(Stats(client))
